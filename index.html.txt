        function setupEventListeners() {

        function populateCategorySelect() {
            const select = document.getElementById('inputCategory');
            select.innerHTML = '';
            CATEGORIES[currentType].forEach(cat => {
                const opt = document.createElement('option');
                opt.value = cat.id;
                opt.textContent = cat.name;
                select.appendChild(opt);
            });
        }

        function updateUI() {
            const selectedDateStr = document.getElementById('filterDate').value;
            if (!selectedDateStr) return;

            const selectedDate = new Date(selectedDateStr);
            const selYear = selectedDate.getFullYear();
            const selMonth = selectedDate.getMonth();

            // Update Badges Text
            document.getElementById('monthBadge').textContent = `เดือน ${thaiMonths[selMonth]}`;
            document.getElementById('yearBadge').textContent = `ปี ${selYear + 543}`;

            // Filter Calculations
            let dayInc = 0, dayExp = 0;
            let monthInc = 0, monthExp = 0;
            let yearInc = 0, yearExp = 0;

            // Expense totals for Top Categories
            const dayCatTotals = {};
            const monthCatTotals = {};
            const yearCatTotals = {};

            transactions.forEach(tx => {
                const txDate = new Date(tx.date);
                const isDay = tx.date === selectedDateStr;
                const isMonth = txDate.getFullYear() === selYear && txDate.getMonth() === selMonth;
                const isYear = txDate.getFullYear() === selYear;

                if (tx.type === 'income') {
                    if (isDay) dayInc += tx.amount;
                    if (isMonth) monthInc += tx.amount;
                    if (isYear) yearInc += tx.amount;
                } else {
                    if (isDay) {
                        dayExp += tx.amount;
                        dayCatTotals[tx.category] = (dayCatTotals[tx.category] || 0) + tx.amount;
                    }
                    if (isMonth) {
                        monthExp += tx.amount;
                        monthCatTotals[tx.category] = (monthCatTotals[tx.category] || 0) + tx.amount;
                    }
                    if (isYear) {
                        yearExp += tx.amount;
                        yearCatTotals[tx.category] = (yearCatTotals[tx.category] || 0) + tx.amount;
                    }
                }
            });

            // Update Cards UI
            document.getElementById('dayBalance').textContent = formatCurrency(dayInc - dayExp);
            document.getElementById('dayIncome').textContent = formatCurrency(dayInc);
            document.getElementById('dayExpense').textContent = formatCurrency(dayExp);

            document.getElementById('monthBalance').textContent = formatCurrency(monthInc - monthExp);
            document.getElementById('monthIncome').textContent = formatCurrency(monthInc);
            document.getElementById('monthExpense').textContent = formatCurrency(monthExp);

            document.getElementById('yearBalance').textContent = formatCurrency(yearInc - yearExp);
            document.getElementById('yearIncome').textContent = formatCurrency(yearInc);
            document.getElementById('yearExpense').textContent = formatCurrency(yearExp);

            // Update Insights (Top Categories)
            updateTopCategoryUI('topDayCat', 'topDayAmount', dayCatTotals);
            updateTopCategoryUI('topMonthCat', 'topMonthAmount', monthCatTotals);
            updateTopCategoryUI('topYearCat', 'topYearAmount', yearCatTotals);

            // Update Percentage List & Chart
            renderPercentagesAndChart(dayCatTotals, dayExp);

            // Render Transaction List
            renderTransactionList(selectedDateStr);

            // Refresh Lucide Icons
            lucide.createIcons();
        }

            let topCatKey = null;
            let maxVal = 0;

            for (const [cat, val] of Object.entries(totalsObj)) {
                if (val > maxVal) {
                    maxVal = val;
                    topCatKey = cat;
                }
            }

            if (topCatKey && maxVal > 0) {
                const catObj = CATEGORIES.expense.find(c => c.id === topCatKey);
                catElem.textContent = catObj ? catObj.name : topCatKey;
                amtElem.textContent = formatCurrency(maxVal);
            } else {
                catElem.textContent = "-";
                amtElem.textContent = "0 บาท";
            }
        }

        function renderPercentagesAndChart(dayCatTotals, totalDayExpense) {
            const listContainer = document.getElementById('categoryPercentageList');
            const emptyText = document.getElementById('chartEmptyText');
            listContainer.innerHTML = '';

            const categoriesPresent = Object.keys(dayCatTotals);

            if (categoriesPresent.length === 0 || totalDayExpense === 0) {
                listContainer.innerHTML = `
                    <div class="text-center py-8 text-slate-400 text-xs">
                        ไม่มีข้อมูลรายจ่ายในวันที่เลือก
                    </div>`;
                emptyText.classList.remove('hidden');
                if (chartInstance) chartInstance.destroy();
                return;
            }

            emptyText.classList.add('hidden');

            const chartLabels = [];
            const chartData = [];
            const chartColors = [];

            // Sort by amount desc
            const sorted = categoriesPresent.sort((a, b) => dayCatTotals[b] - dayCatTotals[a]);

            sorted.forEach(catId => {
                const amount = dayCatTotals[catId];
                const pct = ((amount / totalDayExpense) * 100).toFixed(1);
                const catObj = CATEGORIES.expense.find(c => c.id === catId) || { name: catId, color: '#94a3b8', icon: 'tag' };

                chartLabels.push(catObj.name);
                chartData.push(amount);
                chartColors.push(catObj.color);

                // Progress Bar Item HTML
                const item = document.createElement('div');
                item.className = 'space-y-1';
                item.innerHTML = `
                    <div class="flex justify-between text-xs font-medium">
                        <span class="flex items-center gap-1.5 text-slate-700">
                            <span class="w-2.5 h-2.5 rounded-full inline-block" style="background-color: ${catObj.color}"></span>
                            ${catObj.name}
                        </span>
                        <span class="text-slate-600">${formatCurrency(amount)} (${pct}%)</span>
                    </div>
                    <div class="w-full bg-slate-100 h-2 rounded-full overflow-hidden">
                        <div class="h-full rounded-full transition-all duration-500" style="width: ${pct}%; background-color: ${catObj.color}"></div>
                    </div>
                `;
                listContainer.appendChild(item);
            });

            // Update Doughnut Chart
            const ctx = document.getElementById('expenseChart').getContext('2d');
            if (chartInstance) chartInstance.destroy();

            chartInstance = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: chartLabels,
                    datasets: [{
                        data: chartData,
                        backgroundColor: chartColors,
                        borderWidth: 2,
                        borderColor: '#ffffff'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    const val = context.raw || 0;
                                    const p = ((val / totalDayExpense) * 100).toFixed(1);
                                    return ` ${context.label}: ฿${val.toLocaleString()} (${p}%)`;
                                }
                            }
                        }
                    },
                    cutout: '70%'
                }
            });
        }

        function renderTransactionList(selectedDateStr) {
            const container = document.getElementById('transactionList');
            const countBadge = document.getElementById('recordCount');
            container.innerHTML = '';

            const filtered = transactions.filter(t => t.date === selectedDateStr);
            countBadge.textContent = `${filtered.length} รายการ`;

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="p-8 text-center text-slate-400 text-sm">
                        ไม่มีบันทึกรายการสำหรับวันนี้
                    </div>`;
                return;
            }

            filtered.forEach(tx => {
                const catList = CATEGORIES[tx.type];
                const catObj = catList.find(c => c.id === tx.category) || { name: tx.category, icon: 'circle-dollar-sign', color: '#64748b' };
                const isInc = tx.type === 'income';

                const item = document.createElement('div');
                item.className = 'p-4 flex items-center justify-between hover:bg-slate-50/80 transition-colors';
                item.innerHTML = `
                    <div class="flex items-center gap-3">
                        <div class="w-10 h-10 rounded-xl flex items-center justify-center text-white shrink-0" style="background-color: ${catObj.color}">
                            <i data-lucide="${catObj.icon}" class="w-5 h-5"></i>
                        </div>
                        <div>
                            <div class="font-medium text-slate-800 text-sm">${catObj.name}</div>
                            <div class="text-xs text-slate-400">${tx.note ? tx.note : 'ไม่ได้ระบุหมายเหตุ'}</div>
                        </div>
                    </div>
                    <div class="flex items-center gap-3">
                        <span class="font-bold text-sm ${isInc ? 'text-emerald-600' : 'text-slate-800'}">
                            ${isInc ? '+' : '-'}${formatCurrency(tx.amount)}
                        </span>
                        <button onclick="deleteTransaction('${tx.id}')" class="text-slate-300 hover:text-rose-500 p-1.5 rounded-lg transition-colors">
                            <i data-lucide="trash-2" class="w-4 h-4"></i>
                        </button>
                    </div>
                `;
                container.appendChild(item);
            });
        }

        function deleteTransaction(id) {
            transactions = transactions.filter(t => t.id !== id);
            showToast("ลบรายการเรียบร้อย");
            updateUI();
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            const msgSpan = document.getElementById('toastMsg');
            msgSpan.textContent = message;

            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2500);
        }
    </script>
